  setIsLoading(true)
  setError(null)
}

const onSubmit = async (data: RegisterFormData) => {
  // Prevent multiple submissions
  if (isLoading) return

    }
  }
}

  if (error instanceof TypeError && error.message.includes('fetch')) {

}
